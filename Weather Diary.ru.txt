Разберу задачу пошагово — от создания интерфейса до публикации на GitHub.
import tkinter as tk
from tkinter import ttk, messagebox
import json
import os
from datetime import datetime

# ---------- Файл для сохранения данных ----------
DATA_FILE = "weather_diary.json"

# ---------- Загрузка данных из JSON ----------
def load_data():
    if os.path.exists(DATA_FILE):
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    return []

# ---------- Сохранение данных в JSON ----------
def save_data(data):
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=4, ensure_ascii=False)

# ---------- Обновление таблицы записей ----------
def update_table(records=None):
    # Очищаем таблицу
    for row in tree.get_children():
        tree.delete(row)
    
    # Если записи не переданы, показываем все
    if records is None:
        records = diary_entries
    
    # Добавляем записи в таблицу
    for entry in records:
        tree.insert("", tk.END, values=(
            entry["date"],
            entry["temperature"],
            entry["description"],
            "Да" if entry["precipitation"] else "Нет"
        ))

# ---------- Добавление новой записи ----------
def add_entry():
    # Получаем данные из полей ввода
    date = date_entry.get()
    temperature = temp_entry.get()
    description = desc_entry.get()
    precipitation = precip_var.get()
    
    # Проверка корректности ввода
    try:
        # Проверяем формат даты (ДД.ММ.ГГГГ или ГГГГ-ММ-ДД)
        datetime.strptime(date, "%d.%m.%Y")
    except ValueError:
        try:
            datetime.strptime(date, "%Y-%m-%d")
        except ValueError:
            messagebox.showerror("Ошибка", "Дата должна быть в формате ДД.ММ.ГГГГ или ГГГГ-ММ-ДД")
            return
    
    try:
        temp_value = float(temperature)
    except ValueError:
        messagebox.showerror("Ошибка", "Температура должна быть числом")
        return
    
    if not description.strip():
        messagebox.showerror("Ошибка", "Описание погоды не может быть пустым")
        return
    
    # Создаём запись
    new_entry = {
        "date": date,
        "temperature": temp_value,
        "description": description,
        "precipitation": precipitation
    }
    
    diary_entries.append(new_entry)
    save_data(diary_entries)
    update_table()
    
    # Очищаем поля ввода
    date_entry.delete(0, tk.END)
    temp_entry.delete(0, tk.END)
    desc_entry.delete(0, tk.END)
    precip_var.set(False)
    
    messagebox.showinfo("Успех", "Запись добавлена!")

# ---------- Фильтрация по дате ----------
def filter_by_date():
    filter_date = filter_date_entry.get()
    if not filter_date:
        messagebox.showwarning("Предупреждение", "Введите дату для фильтрации")
        return
    
    try:
        datetime.strptime(filter_date, "%d.%m.%Y")
    except ValueError:
        try:
            datetime.strptime(filter_date, "%Y-%m-%d")
        except ValueError:
            messagebox.showerror("Ошибка", "Дата должна быть в формате ДД.ММ.ГГГГ или ГГГГ-ММ-ДД")
            return
    
    filtered = [entry for entry in diary_entries if entry["date"] == filter_date]
    
    if filtered:
        update_table(filtered)
        messagebox.showinfo("Результат", f"Найдено {len(filtered)} записей")
    else:
        messagebox.showinfo("Результат", "Записей на эту дату не найдено")
        update_table()

# ---------- Фильтрация по температуре ----------
def filter_by_temperature():
    try:
        min_temp = float(filter_temp_min_entry.get()) if filter_temp_min_entry.get() else -float('inf')
        max_temp = float(filter_temp_max_entry.get()) if filter_temp_max_entry.get() else float('inf')
    except ValueError:
        messagebox.showerror("Ошибка", "Температура должна быть числом")
        return
    
    filtered = [entry for entry in diary_entries 
                if min_temp <= entry["temperature"] <= max_temp]
    
    if filtered:
        update_table(filtered)
        messagebox.showinfo("Результат", f"Найдено {len(filtered)} записей")
    else:
        messagebox.showinfo("Результат", "Записей с такой температурой не найдено")
        update_table()

# ---------- Сброс фильтрации (показать все записи) ----------
def reset_filter():
    filter_date_entry.delete(0, tk.END)
    filter_temp_min_entry.delete(0, tk.END)
    filter_temp_max_entry.delete(0, tk.END)
    update_table()
    messagebox.showinfo("Фильтр сброшен", "Показаны все записи")

# ---------- Удаление выбранной записи ----------
def delete_selected():
    selected = tree.selection()
    if not selected:
        messagebox.showwarning("Предупреждение", "Выберите запись для удаления")
        return
    
    if messagebox.askyesno("Подтверждение", "Удалить выбранную запись?"):
        # Получаем индекс выбранной записи
        for item in selected:
            values = tree.item(item)["values"]
            # Ищем запись в списке
            for i, entry in enumerate(diary_entries):
                if (entry["date"] == values[0] and 
                    entry["temperature"] == float(values[1]) and
                    entry["description"] == values[2]):
                    diary_entries.pop(i)
                    break
        
        save_data(diary_entries)
        update_table()
        messagebox.showinfo("Успех", "Запись удалена")

# ---------- Инициализация данных ----------
diary_entries = load_data()

# ========== Панель фильтрации ==========
filter_frame = ttk.LabelFrame(root, text="Фильтрация записей", padding=10)
filter_frame.pack(fill="x", padx=10, pady=5)

# Фильтр по дате
ttk.Label(filter_frame, text="Дата:").grid(row=0, column=0, padx=5, pady=5, sticky="w")

filter_date_entry.grid(row=0, column=1, padx=5, pady=5)

filter_date_btn.grid(row=0, column=2, padx=5, pady=5)

# Фильтр по температуре
ttk.Label(filter_frame, text="Температура от:").grid(row=0, column=3, padx=5, pady=5, sticky="w")
filter_temp_min_entry = ttk.Entry(filter_frame, width=8)
filter_temp_min_entry.grid(row=0, column=4, padx=5, pady=5)

ttk.Label(filter_frame, text="до:").grid(row=0, column=5, padx=5, pady=5, sticky="w")
filter_temp_max_entry = ttk.Entry(filter_frame, width=8)
filter_temp_max_entry.grid(row=0, column=6, padx=5, pady=5)

filter_temp_btn = ttk.Button(filter_frame, text="Фильтр по температуре", command=filter_by_temperature)
filter_temp_btn.grid(row=0, column=7, padx=5, pady=5)

# Кнопка сброса фильтра
reset_btn = ttk.Button(filter_frame, text="Сбросить фильтр", command=reset_filter)
reset_btn.grid(row=0, column=8, padx=10, pady=5)

# ========== Таблица записей ==========

# Создаём таблицу
columns = ("Дата", "Температура", "Описание", "Осадки")
